cmake_minimum_required(VERSION 3.16)
project(v3_moveit)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
find_package(joint_trajectory_controller REQUIRED)
find_package(controller_interface REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(moveit_core REQUIRED)
find_package(ruckig CONFIG REQUIRED)
find_package(moveit_servo REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(control_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)

set(JAKA_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/jaka" CACHE PATH "Local SDK, never downloaded from another version")
if(NOT EXISTS "${JAKA_SDK_ROOT}/include/jakaAPI.h")
  message(FATAL_ERROR "Missing local JAKA SDK. Run tools/prepare_assets.py on the source machine.")
endif()
add_library(v3_jaka_system SHARED cpp/jaka_system.cpp)
target_include_directories(v3_jaka_system PRIVATE include "${JAKA_SDK_ROOT}/include")
ament_target_dependencies(v3_jaka_system rclcpp rclcpp_lifecycle hardware_interface pluginlib std_msgs)
target_link_libraries(v3_jaka_system "${JAKA_SDK_ROOT}/lib/libjakaAPI.so" pthread)
# Fail during the build if an SDK declaration does not match the linked library.
target_link_options(v3_jaka_system PRIVATE "LINKER:--no-undefined")
set_target_properties(v3_jaka_system PROPERTIES INSTALL_RPATH "$ORIGIN/jaka")
pluginlib_export_plugin_description_file(hardware_interface jaka_system.xml)
install(TARGETS v3_jaka_system LIBRARY DESTINATION lib)
install(FILES "${JAKA_SDK_ROOT}/lib/libjakaAPI.so" DESTINATION lib/jaka)
# Test-only final command audit. Keep this library independent of the robot SDK.
add_library(v3_guarded_generic_system SHARED cpp/guarded_generic_system.cpp)
target_include_directories(v3_guarded_generic_system PRIVATE include)
ament_target_dependencies(v3_guarded_generic_system rclcpp rclcpp_lifecycle hardware_interface pluginlib)
target_link_libraries(v3_guarded_generic_system hardware_interface::mock_components)
target_link_options(v3_guarded_generic_system PRIVATE "LINKER:--no-undefined")
pluginlib_export_plugin_description_file(hardware_interface guarded_generic_system.xml)
install(TARGETS v3_guarded_generic_system LIBRARY DESTINATION lib)

add_library(v3_path_ruckig_retiming SHARED cpp/path_ruckig_retiming.cpp)
target_include_directories(v3_path_ruckig_retiming PRIVATE include)
ament_target_dependencies(v3_path_ruckig_retiming moveit_core pluginlib)
target_link_libraries(v3_path_ruckig_retiming ruckig::ruckig)
target_link_options(v3_path_ruckig_retiming PRIVATE "LINKER:--no-undefined")
pluginlib_export_plugin_description_file(moveit_core path_ruckig_retiming.xml)
install(TARGETS v3_path_ruckig_retiming LIBRARY DESTINATION lib)

add_library(v3_smooth_trajectory_controller SHARED cpp/smooth_trajectory_controller.cpp)
target_include_directories(v3_smooth_trajectory_controller PRIVATE include)
ament_target_dependencies(v3_smooth_trajectory_controller joint_trajectory_controller controller_interface rclcpp_action std_msgs pluginlib)
target_link_libraries(v3_smooth_trajectory_controller ruckig::ruckig)
target_link_options(v3_smooth_trajectory_controller PRIVATE "LINKER:--no-undefined")
pluginlib_export_plugin_description_file(controller_interface smooth_trajectory_controller.xml)
install(TARGETS v3_smooth_trajectory_controller LIBRARY DESTINATION lib)

add_executable(v3_controller_activation_tests EXCLUDE_FROM_ALL ../../tools/controller_activation_tests.cpp)
target_include_directories(v3_controller_activation_tests PRIVATE include)
ament_target_dependencies(v3_controller_activation_tests joint_trajectory_controller controller_interface rclcpp_action std_msgs pluginlib)
target_link_libraries(v3_controller_activation_tests ruckig::ruckig)

add_executable(v3_path_retime_tests EXCLUDE_FROM_ALL ../../tools/path_retime_tests.cpp)
target_include_directories(v3_path_retime_tests PRIVATE include)
ament_target_dependencies(v3_path_retime_tests joint_trajectory_controller)
target_link_libraries(v3_path_retime_tests ruckig::ruckig)

add_executable(v3_stop_envelope_tests EXCLUDE_FROM_ALL ../../tools/stop_envelope_tests.cpp)
target_include_directories(v3_stop_envelope_tests PRIVATE include)
ament_target_dependencies(v3_stop_envelope_tests moveit_core pluginlib)
target_link_libraries(v3_stop_envelope_tests ruckig::ruckig)

add_executable(v3_fixed_servo_node cpp/fixed_servo_node.cpp)
target_include_directories(v3_fixed_servo_node PRIVATE include)
ament_target_dependencies(v3_fixed_servo_node rclcpp moveit_servo moveit_ros_planning moveit_core
  control_msgs geometry_msgs std_msgs std_srvs trajectory_msgs moveit_msgs)
target_link_libraries(v3_fixed_servo_node ruckig::ruckig)
target_link_options(v3_fixed_servo_node PRIVATE "LINKER:--no-undefined")
install(TARGETS v3_fixed_servo_node DESTINATION lib/${PROJECT_NAME})

add_executable(v3_servo_interval_tests EXCLUDE_FROM_ALL ../../tools/servo_interval_tests.cpp)
target_include_directories(v3_servo_interval_tests PRIVATE include)
ament_target_dependencies(v3_servo_interval_tests joint_trajectory_controller)
target_link_libraries(v3_servo_interval_tests ruckig::ruckig)

add_executable(v3_edg_feedback_tests EXCLUDE_FROM_ALL ../../tools/edg_feedback_tests.cpp)
target_include_directories(v3_edg_feedback_tests PRIVATE include)

ament_python_install_package(v3_moveit)
install(PROGRAMS scripts/quest_bridge scripts/supervisor scripts/scene_loader scripts/synthetic_quest
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config description DESTINATION share/${PROJECT_NAME})
install(FILES jaka_system.xml DESTINATION share/${PROJECT_NAME})
ament_package()
